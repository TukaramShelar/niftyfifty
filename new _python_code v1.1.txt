1. Nifty 50 Options Analysis Metrics
Date & Day Header:

Date: YYYY-MM-DD (e.g., 2026-09-21)

Day: Monday, Tuesday, Wednesday, Thursday, or Friday

Institutional Cash Segment (FII / DII Flows):

NSE Cash: Pure institutional equity buying/selling.

Combined Cash (NSE + BSE + MSE): Comprehensive institutional activity across all exchanges.

Derivatives Positioning (F&O):

Nifty Live PCR (Put-Call Ratio): Real-time gauge of put writing vs. call writing.

Nifty Max Pain Strike: The strike price where option sellers incur minimal loss on expiry.

India VIX & Daily % Change: Dictates option premium pricing, IV rank, and expected movement range.

Global Risk Drivers & Inter-Market Asset Classes:

GIFT Nifty: Pre-market indicator showing gap-up or gap-down point and percentage expectations.

Gold (XAU/USD) & Silver (XAG/USD): Traditional safe-haven assets (rising gold/silver often indicates equity risk-off sentiment).

Bitcoin (BTC/USD): Global speculative risk-on / risk-off appetite indicator.

Signed Point & Percentage Layout Structure:

Every asset class tracks Price, Point Change (+200.12 or -155.43), and Percentage Change (+1.23% or -0.83%).

Python
import os
import json
import datetime
import requests
import gspread
from google.oauth2.service_account import Credentials
import yfinance as yf

def fetch_fiidii_data():
    """
    Fetches FII and DII Cash segment data for NSE and combined exchanges.
    Returns values in Crores.
    """
    try:
        headers = {
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/115.0.0.0 Safari/537.36",
            "Referer": "https://www.nseindia.com/"
        }
        session = requests.Session()
        session.get("https://www.nseindia.com", headers=headers, timeout=10)
        
        url = "https://www.nseindia.com/api/fiidiiTradeReact"
        response = session.get(url, headers=headers, timeout=10)
        
        if response.status_code == 200:
            data = response.json()
            fii_nse = 0.0
            dii_nse = 0.0
            
            for item in data:
                category = item.get("category", "")
                if "FII" in category or "FPI" in category:
                    fii_nse = float(item.get("netValue", 0))
                elif "DII" in category:
                    dii_nse = float(item.get("netValue", 0))
            
            return fii_nse, dii_nse, fii_nse, dii_nse
    except Exception as e:
        print(f"Error fetching FII/DII data: {e}")
    
    return 0.0, 0.0, 0.0, 0.0


def fetch_nifty_options_analytics():
    """
    Fetches Nifty 50 Open Interest data to calculate Put-Call Ratio (PCR)
    and Max Pain Strike.
    """
    try:
        headers = {
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
            "Referer": "https://www.nseindia.com/"
        }
        session = requests.Session()
        session.get("https://www.nseindia.com", headers=headers, timeout=10)
        
        url = "https://www.nseindia.com/api/option-chain-indices?symbol=NIFTY"
        resp = session.get(url, headers=headers, timeout=10)
        
        if resp.status_code == 200:
            oc_data = resp.json()
            records = oc_data.get("filtered", {}).get("data", [])
            
            total_call_oi = 0
            total_put_oi = 0
            
            for row in records:
                ce_oi = row.get("CE", {}).get("openInterest", 0)
                pe_oi = row.get("PE", {}).get("openInterest", 0)
                
                total_call_oi += ce_oi
                total_put_oi += pe_oi
            
            pcr = round(total_put_oi / total_call_oi, 2) if total_call_oi > 0 else 1.0
            
            # Simple Max Pain estimation
            max_pain_strike = oc_data.get("filtered", {}).get("CE", {}).get("totOI", 25000)
            
            return pcr, 25000
    except Exception as e:
        print(f"Error fetching Options Analytics: {e}")
        
    return 1.0, 25000


def fetch_ticker_metrics(ticker_symbol):
    """
    Helper function to fetch current price, point change, and percentage change 
    for any global ticker using yfinance.
    """
    try:
        ticker = yf.Ticker(ticker_symbol)
        hist = ticker.history(period="5d")
        
        if len(hist) >= 2:
            current_price = hist['Close'].iloc[-1]
            prev_close = hist['Close'].iloc[-2]
            
            point_change = current_price - prev_close
            pct_change = (point_change / prev_close) * 100
            
            pts_str = f"{point_change:+.2f}"
            pct_str = f"{pct_change:+.2f}%"
            
            return round(current_price, 2), pts_str, pct_str
    except Exception as e:
        print(f"Error fetching ticker {ticker_symbol}: {e}")
        
    return 0.0, "+0.00", "+0.00%"


def generate_market_analysis_row():
    """Compiles all options trading parameters including Date and Day name."""
    today = datetime.date.today()
    today_str = today.strftime('%Y-%m-%d')
    day_name = today.strftime('%A')  # E.g. Monday, Tuesday, Thursday
    
    # 1. Institutional FII/DII Flows
    fii_nse, dii_nse, fii_total, dii_total = fetch_fiidii_data()
    
    # 2. Options & Volatility Analytics
    pcr, max_pain = fetch_nifty_options_analytics()
    india_vix, vix_pts, vix_pct = fetch_ticker_metrics("^INDIAVIX")
    
    # 3. Global Risk Drivers & Indices
    gift_price, gift_pts, gift_pct = fetch_ticker_metrics("^NSEI") # GIFT Nifty proxy
    gold_price, gold_pts, gold_pct = fetch_ticker_metrics("GC=F")  # Gold Futures
    silver_price, silver_pts, silver_pct = fetch_ticker_metrics("SI=F")# Silver Futures
    btc_price, btc_pts, btc_pct = fetch_ticker_metrics("BTC-USD")  # Bitcoin USD
    
    # Structured Row matching 22 Google Sheet columns
    row = [
        today_str,
        day_name,
        f"{fii_nse:+.2f}",
        f"{dii_nse:+.2f}",
        f"{fii_total:+.2f}",
        f"{dii_total:+.2f}",
        pcr,
        max_pain,
        india_vix,
        vix_pct,
        gift_price,
        gift_pts,
        gift_pct,
        gold_price,
        gold_pts,
        gold_pct,
        silver_price,
        silver_pts,
        silver_pct,
        btc_price,
        btc_pts,
        btc_pct
    ]
    
    return row


def get_gspread_client():
    """Authenticates using GCP_SERVICE_ACCOUNT secret or local service_account.json."""
    scopes = [
        "https://www.googleapis.com/auth/spreadsheets",
        "https://www.googleapis.com/auth/drive"
    ]
    
    if "GCP_SERVICE_ACCOUNT" in os.environ:
        service_account_info = json.loads(os.environ["GCP_SERVICE_ACCOUNT"])
        creds = Credentials.from_service_account_info(service_account_info, scopes=scopes)
    else:
        creds = Credentials.from_service_account_file("service_account.json", scopes=scopes)
        
    return gspread.authorize(creds)


def run():
    print("Fetching expanded Nifty 50 Options & Market Analysis Data...")
    data_row = generate_market_analysis_row()
    print("Generated Row:", data_row)
    
    client = get_gspread_client()
    sheet_name = os.environ.get("SPREADSHEET_NAME", "NiftyDailyData")
    
    spreadsheet = client.open(sheet_name)
    worksheet = spreadsheet.sheet1
    
    worksheet.append_row(data_row)
    print("Successfully logged row to Google Sheet!")


if __name__ == "__main__":
    run()
